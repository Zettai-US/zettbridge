cmake_minimum_required(VERSION 3.16)
project(zettbridge_llama_agent C CXX)
set(CMAKE_CXX_STANDARD 17)
set(LLAMA_SOURCE "" CACHE PATH "Pinned, unmodified llama.cpp source")
if(NOT EXISTS "${LLAMA_SOURCE}/include/llama.h")
    message(FATAL_ERROR "Set LLAMA_SOURCE to a llama.cpp checkout")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
foreach(feature SSE42 AVX AVX2 FMA F16C BMI2)
    set(GGML_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
# KVM's UC MMIO emulator cannot execute AVX memory operands. Keep this
# functional profile scalar; both the reference and shared paths use it.
add_compile_options(-fno-tree-vectorize -fno-tree-slp-vectorize)
add_subdirectory("${LLAMA_SOURCE}" llama)
add_executable(zb-llama-agent main.cpp kv_buffer.cpp attention.cpp)
add_executable(zb-attention-test attention_test.cpp kv_buffer.cpp attention.cpp)
foreach(target zb-llama-agent zb-attention-test)
    target_include_directories(${target} PRIVATE ../../include)
    target_include_directories(${target} SYSTEM PRIVATE "${LLAMA_SOURCE}/vendor" "${LLAMA_SOURCE}/ggml/src")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(${target} PRIVATE llama)
    target_link_options(${target} PRIVATE -static -Wl,--wrap=ggml_backend_alloc_ctx_tensors_from_buft
        -Wl,--wrap=memcpy -Wl,--wrap=memmove -Wl,--wrap=memset
        -Wl,--wrap=ggml_compute_forward_flash_attn_ext)
endforeach()
enable_testing()
add_test(NAME direct-kv-attention COMMAND zb-attention-test)
